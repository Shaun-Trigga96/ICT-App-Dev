DROP TABLE IF EXISTS Sale;
DROP TABLE IF EXISTS Agent;
DROP TABLE IF EXISTS Property;
DROP TABLE IF EXISTS Owner;

CREATE TABLE IF NOT EXISTS [Agent] (
	"agent_no" INTEGER NOT NULL UNIQUE,
	"name" TEXT,
	"homePhone" INTEGER,
	"cellPhone"	INTEGER,
	"email"	TEXT,
	"annual_sales" REAL,
	PRIMARY KEY("agent_no" AUTOINCREMENT)
);

CREATE TABLE IF NOT EXISTS [Owner] (
	"owner_no" INTEGER UNIQUE,
	"name" TEXT,
	"homePhone" INTEGER,
	"cellPhone" INTEGER,
	"email"	TEXT,
	PRIMARY KEY("owner_no" AUTOINCREMENT)
);

CREATE TABLE IF NOT EXISTS [Property] (
	"property_no"	INTEGER,
	"owner_no"	INTEGER,
	"prop_address" TEXT,
	"year_built" INTEGER,
	"no_bedrooms" INTEGER,
	"no_bathrooms" INTEGER,
	"no_garages" INTEGER,
	"date_listed" TEXT,
	"asking_price" REAL,
	"last_update" TEXT,
	"status" TEXT,
	FOREIGN KEY ("owner_no") REFERENCES Owner("owner_no")
	PRIMARY KEY("property_no" AUTOINCREMENT)
);

CREATE TABLE IF NOT EXISTS [Sale] (
	"property_no" INTEGER,
	"owner_no" INTEGER,
	"agent_no" INTEGER,
	"sale_date"	TEXT,
	"sale_price" REAL,
	FOREIGN KEY("property_no") REFERENCES Property("property_no")
	FOREIGN KEY("owner_no") REFERENCES Owner("owner_no")
	FOREIGN KEY("agent_no") REFERENCES Agent("agent_no")
	PRIMARY KEY("property_no","owner_no","agent_no")
);
