            Log.e("STATE", "Payment Received!!!!!!!!");
            try {
                Connection connection = null;
                connection = connectToDb();
                Log.e("STATE", connection.getCatalog());
                String sqlInsert = "Insert into Payment ('" + Payment_Number.getText().toString() + "','" + Receipt_Number.getText().toString() + "'," +
                        " '" + Payment_Amount.getText().toString() + "','" + Payment_Type.getText().toString() + "'," +
                        "' '" + Payment_Date.getText().toString() + "','" + Appointment_No.getText().toString() + "')";
                Statement st = connection.createStatement();
                ResultSet rs = st.executeQuery(sqlInsert);

            } catch (Exception exception) {
                Log.e("Error", exception.getMessage());
            }
        });

        btnUpdate.setOnClickListener(v -> {
            try {
                Connection connection = null;
                connection = connectToDb();
                if (connection != null) {
                    String sqlDelete = "Update Payment Set Payment_Number ='" + Payment_Number.getText().toString() + "',Receipt_Number ='" + Receipt_Number.getText().toString() + "',Payment_Amount=" +
                            " '" + Payment_Amount.getText().toString() + "',Payment_Type ='" + Payment_Type.getText().toString() + "'Payment_Date =" +
                            "' '" + Payment_Date.getText().toString() + "''" + "',Appointment_No ='" + Appointment_No.getText().toString() + "'where Payment_Number = '" + Payment_Number.getText().toString() + "'";
                    Statement st = connection.createStatement();
                    ResultSet rs = st.executeQuery(sqlDelete);

                }
            } catch (Exception exception) {
                Log.e("Error", exception.getMessage());
            }
        });

        btnDelete.setOnClickListener(v -> {
            try {
                Connection connection = null;
                connection = connectToDb();
                if (connection != null) {
                    String sqlDelete = "Delete Payment where Payment_Number ='" + Payment_Number.getText().toString() +"'";
                    Statement st = connection.createStatement();
                    ResultSet rs = st.executeQuery(sqlDelete);

                }
            } catch (Exception exception) {
                Log.e("Error", exception.getMessage());
            }
        });

        btnDisplay.setOnClickListener(v -> {
            try {
                Connection connection = null;
                connection = connectToDb();
                if (connection != null) {
                    String sqlGet = "Select * from Payment where Payment_Number ='" + Payment_Number.getText().toString() +"'";
                    Statement st = connection.createStatement();
                    ResultSet rs = st.executeQuery(sqlGet);

                    while(rs.next()){
                    Payment_Number.setText(rs.getString(2));
                    Receipt_Number.setText(rs.getString(3));
                    Payment_Amount.setText(rs.getString(4));
                    Payment_Type.setText(rs.getString(5));
                    Payment_Date.setText(rs.getString(6));
                    Appointment_No.setText(rs.getString(7));
                    }
                }
            } catch (Exception exception) {
                Log.e("Error", exception.getMessage());
            }
        });

    }
